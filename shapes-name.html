<script>
/* ---- データ ---- */
const SHAPES = [
{ jp:'円', en:'circle', kana:'サークル', inner:'<circle class="shape" cx="60" cy="50" r="38"/>' },
{ jp:'三角形', en:'triangle', kana:'トライアングル', inner:'<polygon class="shape" points="60,14 96,78 24,78"/>' },
{ jp:'四角形', en:'square', kana:'スクエア', inner:'<polygon class="shape" points="30,16 90,16 90,76 30,76"/>' },
{ jp:'五角形', en:'pentagon', kana:'ペンタゴン', inner:'<polygon class="shape" points="60,12 98,40 84,84 36,84 22,40"/>' },
{ jp:'六角形', en:'hexagon', kana:'ヘキサゴン', inner:'<polygon class="shape" points="80,15 100,50 80,85 40,85 20,50 40,15"/>' },
{ jp:'八角形', en:'octagon', kana:'オクタゴン', inner:'<polygon class="shape" points="45,13 75,13 97,35 97,65 75,87 45,87 23,65 23,35"/>' },
];
const TOTAL = 10;
function svg(shape){
return '<svg viewBox="0 0 120 100" xmlns="http://www.w3.org/2000/svg" aria-hidden="true">'+shape.inner+'</svg>';
}
function shuffle(a){ a=a.slice(); for(let i=a.length-1;i>0;i--){const j=Math.floor(Math.random()*(i+1));[a[i],a[j]]=[a[j],a[i]];} return a; }
function pickOptions(correct){
const others = shuffle(SHAPES.filter(s=>s!==correct)).slice(0,3);
return shuffle([correct, ...others]);
}
let mode='mix', qIndex=0, score=0, answered=false;
const $ = id => document.getElementById(id);
function show(sec){ ['start','quiz','result'].forEach(s=>$(s).classList.toggle('hidden', s!==sec)); }
function goStart(){ show('start'); }
function startQuiz(m){ mode=m; qIndex=0; score=0; $('qtotal').textContent=TOTAL; show('quiz'); nextQuestion(); }
function restart(){ startQuiz(mode); }
function currentDirection(){
if(mode==='mix') return Math.random()<0.5 ? 'word2shape':'shape2word';
return mode;
}
function nextQuestion(){
answered=false;
$('feedback').innerHTML='';
$('qnum').textContent=qIndex+1;
$('score').textContent=score;
const dir=currentDirection();
const correct=SHAPES[Math.floor(Math.random()*SHAPES.length)];
const opts=pickOptions(correct);
const prompt=$('prompt'), options=$('options');
options.innerHTML='';
if(dir==='word2shape'){
prompt.innerHTML=
'<p class="ask">この ことばの かたちは？</p>'
+'<div class="word-kana">'+correct.kana+'</div>'
+'<div class="word-en">'+correct.en+'</div>';
opts.forEach(s=>{
const b=document.createElement('button');
b.className='opt'; b.dataset.en=s.en;
b.innerHTML=svg(s);
b.onclick=()=>choose(b, s, correct);
options.appendChild(b);
});
}else{
prompt.innerHTML=
'<p class="ask">この かたちの ことばは？</p>'
+'<div class="big-shape">'+svg(correct)+'</div>';
opts.forEach(s=>{
const b=document.createElement('button');
b.className='opt'; b.dataset.en=s.en;
b.innerHTML='<div class="w-kana">'+s.kana+'</div><div class="w-en">'+s.en+'</div>';
b.onclick=()=>choose(b, s, correct);
options.appendChild(b);
});
}
}
function choose(btn, chosen, correct){
if(answered) return;
answered=true;
const isCorrect = chosen.en===correct.en;
const all=[...document.querySelectorAll('#options .opt')];
all.forEach(o=>{
o.classList.add('locked');
if(o.dataset.en===correct.en){
o.classList.add('correct');
o.insertAdjacentHTML('beforeend','<div class="mark">○</div>');
}
});
if(isCorrect){ score++; $('score').textContent=score; }
else{ btn.classList.add('wrong'); btn.insertAdjacentHTML('beforeend','<div class="mark">✕</div>'); }
const fb=$('feedback');
fb.innerHTML=
'<p class="fb-msg '+(isCorrect?'ok':'ng')+'">'+(isCorrect?'せいかい！':'ざんねん')+'</p>'
+'<p class="fb-ans">こたえ： <b>'+correct.kana+' / '+correct.en+'</b>（'+correct.jp+'）</p>'
+'<button class="btn btn-primary" id="nextBtn">'+(qIndex+1>=TOTAL?'けっかを見る':'つぎの もんだい')+'</button>';
$('nextBtn').onclick=()=>{ qIndex++; if(qIndex>=TOTAL) showResult(); else nextQuestion(); };
$('nextBtn').focus();
}
function showResult(){
show('result');
$('finalScore').textContent=score;
$('finalTotal').textContent=TOTAL;
const ratio=score/TOTAL;
const filled=Math.round(ratio*5);
$('stars').textContent='★'.repeat(filled)+'☆'.repeat(5-filled);
let msg;
if(ratio===1) msg='ぜんもん せいかい！ すばらしい！';
else if(ratio>=0.7) msg='よくできました！';
else if(ratio>=0.4) msg='その調子。もう一度ちょうせん！';
else msg='くりかえすと おぼえられます。';
$('resultMsg').textContent=msg;
}
$('hero').innerHTML=SHAPES.map(svg).join('');
</script>
